DelegateBasedCRUDRepo

class DelegateBasedCRUDRepo<ObjectType, IdType, InputValueType>(readDelegate: ReadCRUDRepo<ObjectType, IdType>, writeDelegate: WriteCRUDRepo<ObjectType, IdType, InputValueType>) : CRUDRepo<ObjectType, IdType, InputValueType> , ReadCRUDRepo<ObjectType, IdType> , WriteCRUDRepo<ObjectType, IdType, InputValueType> (source)
